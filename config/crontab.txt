# Cron jobs para SETEX prod
# Instalado en root crontab del HOST (no en contenedor setex).
# Reinstalar con: sudo crontab /opt/setex/prod/config/crontab.txt
# Última actualización: 2026-04-21 (post-cutover Fase 4 + fix watchdog)

# SETEX watchdog — cada 5 minutos (auto-healing servicios setex-prod-*)
*/5 * * * * /opt/setex/prod/scripts/watchdog.sh >> /opt/setex/prod/logs/watchdog.log 2>&1

# SETEX fix-permissions — cada hora (red de seguridad)
0 * * * * /opt/setex/prod/scripts/fix-permissions.sh >> /opt/setex/prod/logs/permissions.log 2>&1

# Backup diario de PostgreSQL cifrado GPG a las 03:00 UTC
0 3 * * * /opt/setex/prod/scripts/backup-postgres.sh >> /opt/setex/prod/logs/backup.log 2>&1

# Smoke test OCR — 04:30 UTC diario (06:30 Madrid verano / 05:30 invierno)
# Detecta regresiones de OpenAI schema strict y caídas de Azure DI
30 4 * * * SETEX_OCR_LOG=/opt/setex/prod/logs/smoke-ocr.log /usr/bin/node /opt/setex/prod/scripts/smoke-test-ocr.js

# Replicación offsite de backups a VPS secundario (72.62.189.27) — 05:00 UTC
0 5 * * * /opt/setex/prod/scripts/backup-offsite-replicate.sh >> /opt/setex/prod/logs/backup-offsite.log 2>&1
